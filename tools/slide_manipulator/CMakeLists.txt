cmake_minimum_required(VERSION 3.22)
project(easypath_slide_manipulator LANGUAGES CXX)
find_package(FLTK 1.4 CONFIG REQUIRED)
add_library(manipulator_ui SHARED manipulator.cpp)
target_compile_features(manipulator_ui PRIVATE cxx_std_17)
target_link_libraries(manipulator_ui PRIVATE fltk::fltk)
if(WIN32)
  set_target_properties(manipulator_ui PROPERTIES PREFIX "")
endif()
